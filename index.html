========================= # VARIÁVEIS # ========================= flash_on = False settings_menu = None # ========================= # FRAME PRINCIPAL # ========================= camera_frame = Frame(root, bg="gray15") camera_frame.pack(fill="both", expand=True) # ========================= # MENU CONFIGURAÇÕES # ========================= def toggle_settings(): global settings_menu # FECHAR MENU if settings_menu and settings_menu.winfo_exists(): settings_menu.destroy() return # ABRIR MENU settings_menu = Frame( camera_frame, bg="#1f1f1f", bd=2, relief="ridge" ) settings_menu.place(x=240, y=60) Label( settings_menu, text="Configurações", bg="#1f1f1f", fg="white", font=("Arial", 12, "bold") ).pack(pady=10) options = [ "📷 Grade", "🌙 Modo Noturno", "🎥 Estabilização", "✨ HDR", "🔊 Som da câmera", "📍 Localização" ] for item in options: Button( settings_menu, text=item, anchor="w", width=22, bg="#2b2b2b", fg="white", activebackground="#444", activeforeground="white", relief="flat", padx=10, pady=8 ).pack(fill="x", padx=10, pady=3) # ========================= # FLASH # ========================= def toggle_flash(): global flash_on flash_on = not flash_on if flash_on: flash_button.config( text="⚡", fg="yellow" ) else: flash_button.config( text="⚡̶", fg="white" ) # ========================= # MUDAR MODO # ========================= def change_mode(mode): # RESETAR BOTÕES for m, btn in mode_buttons.items(): btn.config(fg="white") # DESTACAR SELECIONADO mode_buttons[mode].config(fg="yellow") # FOTO if mode == "FOTO": quality_label.config( text="12MP" ) capture_btn.itemconfig( capture_circle, fill="white" ) # VÍDEO elif mode == "VÍDEO": quality_label.config( text="4K 24FPS" ) capture_btn.itemconfig( capture_circle, fill="red" ) # CÂMERA LENTA elif mode == "CÂMERA LENTA": quality_label.config( text="1080P 240FPS" ) capture_btn.itemconfig( capture_circle, fill="red" ) # CINEMA elif mode == "CINEMA": quality_label.config( text="4K 60FPS" ) capture_btn.itemconfig( capture_circle, fill="red" ) # RETRATO elif mode == "RETRATO": quality_label.config( text="12MP" ) capture_btn.itemconfig( capture_circle, fill="white" ) # ========================= # TOPO # ========================= top_bar = Frame(camera_frame, bg="gray15") top_bar.pack(fill="x", pady=15, padx=10) # ESQUERDA left_frame = Frame(top_bar, bg="gray15") left_frame.pack(side="left") quality_label = Label( left_frame, text="12MP", fg="white", bg="gray30", padx=10, pady=5, font=("Arial", 10, "bold") ) quality_label.pack(side="left") # CENTRO (FLASH) center_frame = Frame(top_bar, bg="gray15") center_frame.pack(side="left", expand=True) flash_button = Button( center_frame, text="⚡̶", command=toggle_flash, font=("Arial", 16), bg="gray15", fg="white", relief="flat", activebackground="gray15", borderwidth=0 ) flash_button.pack() # DIREITA (BOTÕES INVERTIDOS) right_frame = Frame(top_bar, bg="gray15") right_frame.pack(side="right") # ENGRENAGEM PRIMEIRO gear_button = Button( right_frame, text="⚙", command=toggle_settings, font=("Arial", 16), bg="gray15", fg="white", relief="flat", activebackground="gray15", borderwidth=0 ) gear_button.pack(side="right", padx=5) # TIMER DEPOIS timer_button = Button( right_frame, text="⏱", font=("Arial", 16), bg="gray15", fg="white", relief="flat", activebackground="gray15", borderwidth=0 ) timer_button.pack(side="right", padx=5) # ========================= # ÁREA CENTRAL # ========================= center_area = Frame(camera_frame, bg="gray25") center_area.pack(expand=True) focus_square = Frame( center_area, width=80, height=80, bg="gray25", highlightbackground="white", highlightthickness=2 ) focus_square.place( relx=0.5, rely=0.5, anchor="center" ) # ========================= # BOTÃO CAPTURA # ========================= capture_area = Frame(root, bg="black") capture_area.pack(side="bottom", pady=20) capture_btn = Canvas( capture_area, width=90, height=90, bg="black", highlightthickness=0 ) capture_circle = capture_btn.create_oval( 10, 10, 80, 80, outline="white", width=5, fill="white" ) capture_btn.pack() # ========================= # MODOS # ========================= modes_frame = Frame(root, bg="black") modes_frame.pack(side="bottom", fill="x", pady=(0, 15)) modes = [ "RETRATO", "VÍDEO", "FOTO", "CÂMERA LENTA", "CINEMA" ] mode_buttons = {} for mode in modes: color = "yellow" if mode == "FOTO" else "white" btn = Button( modes_frame, text=mode, fg=color, bg="black", activebackground="black", activeforeground="yellow", relief="flat", borderwidth=0, font=("Arial", 10, "bold"), command=lambda m=mode: change_mode(m) ) btn.pack(side="left", expand=True) mode_buttons[mode] = btn # ========================= # INICIAR # ========================= root.mainloop()
